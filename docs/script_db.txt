-- =============================================================================
-- SCRIPT DE CREACIÓN DE BASE DE DATOS - "system-gym"
-- =============================================================================

-- 1. CREACIÓN DEL ESQUEMA
CREATE SCHEMA IF NOT EXISTS system_gym_db;

-- =============================================================================
-- TABLA: tb_roles
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_roles CASCADE;

CREATE TABLE system_gym_db.tb_roles (
    id_rol INT GENERATED ALWAYS AS IDENTITY,
    nombre VARCHAR(50) NOT NULL,
    descripcion VARCHAR(200) NULL,
    CONSTRAINT pk_roles PRIMARY KEY (id_rol)
);

-- =============================================================================
-- TABLA: tb_usuario
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_usuario CASCADE;

CREATE TABLE system_gym_db.tb_usuario (
    id_usuario INT GENERATED ALWAYS AS IDENTITY,
    id_rol INT NOT NULL,
    usuario VARCHAR(50) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    correo VARCHAR(255) NOT NULL,
    dos_pasos_activo BOOLEAN NOT NULL DEFAULT FALSE,
    intentos_fallidos INT NOT NULL DEFAULT 0,
    bloqueado_hasta TIMESTAMP(0) NULL,
    ultimo_acceso TIMESTAMP(0) NULL,
    CONSTRAINT pk_usuario PRIMARY KEY (id_usuario),
    CONSTRAINT uq_usuario_username UNIQUE (usuario),
    CONSTRAINT uq_usuario_correo UNIQUE (correo),
    CONSTRAINT fk_usuarios_roles FOREIGN KEY (id_rol)
        REFERENCES system_gym_db
    .tb_roles (id_rol)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- =============================================================================
-- TABLA: tb_configuracion
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_configuracion CASCADE;

CREATE TABLE system_gym_db.tb_configuracion (
    id_config INT GENERATED ALWAYS AS IDENTITY,
    nombre VARCHAR(100) NOT NULL,
    direccion VARCHAR(255) NOT NULL,
    capacidad_maxima SMALLINT NOT NULL,
    tiempo_sesion SMALLINT NOT NULL DEFAULT 30, -- en minutos para cierre de sesión
    telefono VARCHAR(45) NOT NULL,
    correo VARCHAR(254) NOT NULL,
    ruc VARCHAR(20) NOT NULL,
    CONSTRAINT pk_configuracion PRIMARY KEY (id_config),
    CONSTRAINT uq_configuracion_ruc UNIQUE (ruc)
);

-- =============================================================================
-- TABLA: tb_cliente
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_cliente CASCADE;

CREATE TABLE system_gym_db.tb_cliente (
    id_cliente INT GENERATED ALWAYS AS IDENTITY,
    nombre VARCHAR(45) NOT NULL,
    apellido VARCHAR(45) NOT NULL,
    telefono VARCHAR(20) NOT NULL,
    documento VARCHAR(15) NOT NULL,
    correo VARCHAR(254) NOT NULL,
    fecha_registro DATE NOT NULL DEFAULT CURRENT_DATE,
    sexo SMALLINT NOT NULL, -- 0: Masculino, 1: Femenino, u otros códigos parametrizados
    direccion VARCHAR(255) NOT NULL,
    ciudad VARCHAR(100) NULL,
    contacto_emergencia VARCHAR(45) NULL,
    telefono_emergencia VARCHAR(20) NULL,
    ocupacion VARCHAR(100) NULL,
    fotografia VARCHAR(255) NULL, -- Corregido: Se asignó tipo VARCHAR(255)
    observacion TEXT NULL,
    CONSTRAINT pk_cliente PRIMARY KEY (id_cliente),
    CONSTRAINT uq_cliente_documento UNIQUE (documento),
    CONSTRAINT uq_cliente_correo UNIQUE (correo)
);

-- =============================================================================
-- TABLA: tb_planes_membresia
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_planes_membresia CASCADE;

CREATE TABLE system_gym_db.tb_planes_membresia (
    id_plan INT GENERATED ALWAYS AS IDENTITY,
    nombre VARCHAR(100) NOT NULL,
    duracion_dias INT NOT NULL,
    precio DECIMAL(10,2) NOT NULL,
    acceso_vip BOOLEAN NOT NULL DEFAULT FALSE,
    descripcion TEXT NULL,
    estado SMALLINT NOT NULL DEFAULT 1, -- 1: Activo, 0: Inactivo para ventas
    CONSTRAINT pk_planes_membresia PRIMARY KEY (id_plan)
);

-- =============================================================================
-- TABLA: tb_membresias_clientes
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_membresias_clientes CASCADE;

CREATE TABLE system_gym_db.tb_membresias_clientes (
    id_membresia INT GENERATED ALWAYS AS IDENTITY,
    id_cliente INT NOT NULL,
    id_plan INT NOT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_vencimiento DATE NOT NULL,
    estado SMALLINT NOT NULL, -- Ej: 1 (Activa), 2 (Vencida), 3 (Congelada)
    precio_pagado DECIMAL(10,2) NULL,
    renovacion_automatica BOOLEAN NOT NULL DEFAULT FALSE,
    observaciones TEXT NULL,
    CONSTRAINT pk_membresias_clientes PRIMARY KEY (id_membresia),
    CONSTRAINT fk_membresias_miembro FOREIGN KEY (id_cliente)
        REFERENCES system_gym_db
    .tb_cliente (id_cliente)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_membresias_plan FOREIGN KEY (id_plan)
        REFERENCES system_gym_db
    .tb_planes_membresia (id_plan)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- =============================================================================
-- TABLA: tb_pago
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_pago CASCADE;

CREATE TABLE system_gym_db.tb_pago (
    id_pago INT GENERATED ALWAYS AS IDENTITY,
    id_cliente INT NOT NULL,
    id_membresia INT NULL, -- Corregido: Se hizo nullable para permitir 'ON DELETE SET NULL'
    monto_total DECIMAL(10,2) NOT NULL,
    metodo_pago SMALLINT NOT NULL, -- 1: Efectivo, 2: Tarjeta, 3: Transferencia
    fecha_pago TIMESTAMP(0) NOT NULL DEFAULT CURRENT_TIMESTAMP,
    descuento_aplicado DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    CONSTRAINT pk_pago PRIMARY KEY (id_pago),
    CONSTRAINT fk_pagos_miembro FOREIGN KEY (id_cliente)
        REFERENCES system_gym_db
    .tb_cliente (id_cliente)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_pagos_membresia FOREIGN KEY (id_membresia)
        REFERENCES system_gym_db
    .tb_membresias_clientes (id_membresia)
        ON DELETE SET NULL ON UPDATE CASCADE
);

-- =============================================================================
-- TABLA: tb_ficha_medica
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_ficha_medica CASCADE;

CREATE TABLE system_gym_db.tb_ficha_medica (
    id_evaluacion INT GENERATED ALWAYS AS IDENTITY,
    id_cliente INT NOT NULL,
    id_entrenador INT NOT NULL,
    fecha DATE NOT NULL DEFAULT CURRENT_DATE,
    peso DECIMAL(5,2) NOT NULL,
    altura DECIMAL(5,2) NOT NULL,
    imc DECIMAL(5,2) NOT NULL, -- Corregido: 'mic' cambiado a 'imc'
    porcentaje_grasa DECIMAL(5,2) NOT NULL,
    masa_corporal DECIMAL(5,2) NOT NULL,
    presion_arterial DECIMAL(5,2) NOT NULL,
    frecuencia_cardiaca INT NOT NULL,
    enfermedades TEXT NOT NULL,
    comorbilidades TEXT NOT NULL, -- Corregido: 'comorbidades' a 'comorbilidades'
    alergias TEXT NOT NULL,
    medicamentos TEXT NOT NULL,
    cirugias_previas TEXT NOT NULL,
    lesiones TEXT NOT NULL,
    apto_entrenar BOOLEAN NOT NULL DEFAULT TRUE, -- Estandarizado a Boolean para comodidad en JPA
    observacion TEXT NULL,
    CONSTRAINT pk_ficha_medica PRIMARY KEY (id_evaluacion),
    CONSTRAINT fk_evaluaciones_miembro FOREIGN KEY (id_cliente)
        REFERENCES system_gym_db
    .tb_cliente (id_cliente)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_evaluaciones_entrenador FOREIGN KEY (id_entrenador)
        REFERENCES system_gym_db
    .tb_usuario (id_usuario)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- =============================================================================
-- TABLA: tb_ficha_entrenamiento
-- =============================================================================
DROP TABLE IF EXISTS system_gym_db.tb_ficha_entrenamiento CASCADE;

CREATE TABLE system_gym_db.tb_ficha_entrenamiento (
    id_ficha INT GENERATED ALWAYS AS IDENTITY,
    id_cliente INT NOT NULL,
    id_entrenador INT NOT NULL,
    fecha_creacion DATE NOT NULL DEFAULT CURRENT_DATE,
    objetivo VARCHAR(100) NULL,
    rutina_lunes TEXT NOT NULL,
    rutina_martes TEXT NOT NULL,
    rutina_miercoles TEXT NOT NULL,
    rutina_jueves TEXT NOT NULL,
    rutina_viernes TEXT NOT NULL,
    rutina_sabado TEXT NOT NULL,
    CONSTRAINT pk_ficha_entrenamiento PRIMARY KEY (id_ficha),
    CONSTRAINT fk_fichas_miembro FOREIGN KEY (id_cliente)
        REFERENCES system_gym_db
    .tb_cliente (id_cliente)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_fichas_entrenador FOREIGN KEY (id_entrenador)
        REFERENCES system_gym_db
    .tb_usuario (id_usuario)
        ON DELETE RESTRICT ON UPDATE CASCADE
);

-- =============================================================================
-- CREACIÓN DE ÍNDICES ADICIONALES (Optimización de Consultas)
-- =============================================================================

-- Índices para la tabla tb_usuario
CREATE INDEX idx_usuarios_rol ON system_gym_db.tb_usuario (id_rol);

-- Índices para la tabla tb_membresias_clientes
CREATE INDEX idx_membresias_cliente ON system_gym_db.tb_membresias_clientes (id_cliente);
CREATE INDEX idx_membresias_plan ON system_gym_db.tb_membresias_clientes (id_plan);

-- Índices para la tabla tb_pago
CREATE INDEX idx_pagos_cliente ON system_gym_db.tb_pago (id_cliente);
CREATE INDEX idx_pagos_membresia ON system_gym_db.tb_pago (id_membresia);

-- Índices para la tabla tb_ficha_medica
CREATE INDEX idx_evaluaciones_cliente ON system_gym_db.tb_ficha_medica (id_cliente);
CREATE INDEX idx_evaluaciones_entrenador ON system_gym_db.tb_ficha_medica (id_entrenador);

-- Índices para la tabla tb_ficha_entrenamiento
CREATE INDEX idx_fichas_cliente ON system_gym_db.tb_ficha_entrenamiento (id_cliente);
CREATE INDEX idx_fichas_entrenador ON system_gym_db.tb_ficha_entrenamiento (id_entrenador);
